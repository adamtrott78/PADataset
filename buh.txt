# --- 0) run the orchestrator as a user service that survives SSH disconnect ---
cd ~/adamArchives/Adam/varMax/PADataset
mkdir -p results/buh_logs results/resplice_workers results/bank_workers results/cache_workers
chmod +x ./buh_orchestrate.sh

# run as systemd transient *service*
systemd-run --user --unit=buh_orch --collect --property=WorkingDirectory="$PWD" \
  bash -lc './buh_orchestrate.sh'

# --- 1) system logs (everything) ---
journalctl --user -u buh_orch.service -f

# --- 2) full orchestrator log + important-only view ---
LOG=$(ls -t results/buh_logs/orch_*.log | head -n 1)
tail -f "$LOG"

# important-only stream (filters out “NEXT SEARCH / trying center” spam)
tail -f "$LOG" | grep --line-buffered -E \
'^(===|--- CAPTURE|CAPTURE PASS|CAPTURE FAIL|TXRX GATE|TXRX DONE|RESP SIMPLE DONE|HDR DONE|BANK (DISCOVER|TARGET|ASSEMBLE|WRITE|DONE)|Cache build complete|Cache files present|ERROR|!!!)'

# --- 3) resplice UI (your existing log-based TUI) ---
# (run this while resplice is in progress)
python ./tui_resplice_watch.py

# --- 4) live memory/cpu for MATLAB workers ---
watch -n 2 "ps -C MATLAB -o pid,%cpu,%mem,rss,vsz,etime,cmd --sort=-rss | head -n 25"

# --- 5) per-shard logs (resplice/bank/cache) ---
ls -lh results/resplice_workers | tail
ls -lh results/bank_workers | tail
ls -lh results/cache_workers | tail
